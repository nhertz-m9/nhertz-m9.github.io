【Rust】AsyncFnMut を GATs で強引に実装してみる

May 5, 2026

let mut cx = std::task::Context::from_waker(std::task::Waker::noop());
let _ = std::pin::pin!(_async_block_closure_test()).poll(&mut cx);

async fn _async_block_closure_test() {

    trait LendingFnMut<Arg> {
        type LendingOutput<'t>
        where
            Self: 't;

        fn call_mut<'t>(&'t mut self, arg: Arg) -> Self::LendingOutput<'t>;
    }

    struct AnonymousAsyncClosure<'a> {
        // captures
        xs: &'a mut Vec<i32>
    }

    impl<'a> LendingFnMut<i32> for AnonymousAsyncClosure<'a> {
        
        // impl trait in assoc type now requires nightly...
        
        type LendingOutput<'t> = impl Future<Output = ()> where Self: 't;

        fn call_mut<'t>(&'t mut self, arg: i32) -> Self::LendingOutput<'t> {
            async move {
                self.xs.push(arg);
            }
        }
    }


    let mut xs = vec![];

    let mut cls = AnonymousAsyncClosure { xs: &mut xs };
    cls.call_mut(42).await;

    println!("{xs:?}");
}
