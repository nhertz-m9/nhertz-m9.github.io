#[derive(Default, Clone)]
struct PollLock {
    inner: std::rc::Rc<std::cell::RefCell<bool>>,
}

#[derive(PartialEq, Eq, Debug)]
pub enum CoroutineState {
    Yielded,
    Complete,
}

impl std::future::Future for PollLock {
    type Output = ();

    fn poll(
        self: std::pin::Pin<&mut Self>, 
        _cx: &mut std::task::Context<'_>
    ) -> std::task::Poll<Self::Output> {
        
        if self.inner.borrow().to_owned() {
            std::task::Poll::Pending
        } else {
            std::task::Poll::Ready(())
        }
    }
}

pub struct CoroutineContext {
    lock: PollLock,
}

impl CoroutineContext {
    pub fn r#yield(&mut self) -> impl std::future::Future<Output = ()> {
        *self.lock.inner.borrow_mut() = true;
        self.lock.clone()
    }
}

pub struct Coroutine<F: std::future::Future> {
    task: std::pin::Pin<Box<F>>,
    lock: PollLock,
}

impl<F: std::future::Future> Coroutine<F> {
    pub fn new<T: FnOnce(CoroutineContext) -> F>(task: T) -> Self {
        let lock = PollLock::default();
        Self {
            task: Box::pin(task(CoroutineContext { lock: lock.clone() })),
            lock: lock.clone(),
        }
    }

    pub fn resume(&mut self) -> CoroutineState {
        let mut cx = std::task::Context::from_waker(std::task::Waker::noop());
        *self.lock.inner.borrow_mut() = false;

        match self.task.as_mut().poll(&mut cx) {
            std::task::Poll::Ready(_) => CoroutineState::Complete,
            std::task::Poll::Pending => CoroutineState::Yielded,
        }
    }
}


#[cfg(test)]
mod tests {
    use crate::*;

    #[test]
    fn test() {

        let label = String::from("#");

        let mut coroutine1 = Coroutine::new(async |mut cc| {
            for i in (0..).filter(|i| *i % 2 == 0) {
                println!("{label} {i}");
                cc.r#yield().await;
            }
        });

        let mut coroutine2 = Coroutine::new(async |mut cc| {
            for i in (0..).filter(|i| *i % 2 == 1) {
                println!("{label} {i}");
                cc.r#yield().await;
            }
        });

        coroutine1.resume();
        coroutine2.resume();
        coroutine1.resume();
        coroutine2.resume();
        coroutine1.resume();
        coroutine2.resume();
    }
}
